@app.get("/healthz")
def healthz():
    path = pathlib.Path(store_path).resolve()
    repo = pathlib.Path(__file__).resolve().parents[1]
    try:
        rel = path.relative_to(repo)
    except ValueError:
        rel = path.name
    return jsonify(ok=True, store_path=str(rel))
